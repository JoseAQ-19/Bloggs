{{/* GetOptimizedImage — WPO Partial (Fenix V3) Takes a page context and returns an optimized tag with: - WebP conversion via Hugo image processing - Responsive srcset (400w, 800w, 1200w) - Native lazy loading + async decoding - Proper width/height for CLS prevention Falls back to original URL if image processing is not possible (e.g., external URLs or images not in page bundle / assets). Usage: {{ partial "func/GetOptimizedImage.html" (dict "page" . "class" "img br3" "alt" .Title "eager" false) }} */}} {{- $page := .page -}} {{- $class := .class | default "img" -}} {{- $alt := .alt | default "image" -}} {{- $eager := .eager | default false -}} {{- $loading := cond $eager "eager" "lazy" -}} {{- $sizes := .sizes | default "(max-width: 768px) 100vw, 800px" -}} {{/* Get the featured image path */}} {{- $featured_image := partial "func/GetFeaturedImage.html" $page -}} {{- if $featured_image -}} {{/* Try to process the image via Hugo's image pipeline */}} {{- $imgResource := false -}} {{/* First: try page resources (page bundles) */}} {{- with $page.Params.featured_image -}} {{- with $page.Resources.GetMatch . -}} {{- $imgResource = . -}} {{- end -}} {{- end -}} {{/* Second: try global assets */}} {{- if not $imgResource -}} {{- $path := $page.Params.featured_image | default "" -}} {{- $cleanPath := strings.TrimPrefix "/" $path -}} {{- with resources.Get $cleanPath -}} {{- $imgResource = . -}} {{- end -}} {{- end -}} {{- if $imgResource -}} {{/* SUCCESS: We have a processable image resource */}} {{- $img400 := $imgResource.Resize "400x webp q75" -}} {{- $img800 := $imgResource.Resize "800x webp q80" -}} {{- $img1200 := $imgResource.Resize "1200x webp q85" -}} {{- else -}} {{/* FALLBACK: External URL or non-processable image — use original with lazy loading */}} {{- end -}} {{- end -}}
