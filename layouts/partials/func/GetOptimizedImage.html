{{/* GetOptimizedImage — WPO Partial (Fenix V3) Takes a dict with: - page: the page context - class: CSS classes for the img tag - alt: alt text - eager: bool, if true uses eager loading (for above-fold images) Generates WebP with responsive srcset when Hugo can process the image. Falls back to original URL with lazy loading when not processable. */}} {{- $page := .page -}} {{- $class := .class | default "img" -}} {{- $alt := .alt | default "image" -}} {{- $eager := .eager | default false -}} {{- $loading := cond $eager "eager" "lazy" -}} {{- $sizes := .sizes | default "(max-width: 768px) 100vw, 800px" -}} {{/* Get the featured image URL */}} {{- $featured_image := partial "func/GetFeaturedImage.html" $page -}} {{- if $featured_image -}} {{/* Try to get a processable Hugo image resource */}} {{- $processed := false -}} {{/* Attempt 1: Page bundle resource match */}} {{- with $page.Params.featured_image -}} {{- with $page.Resources.GetMatch . -}} {{- $img400 := .Resize "400x webp q75" -}} {{- $img800 := .Resize "800x webp q80" -}} {{- $img1200 := .Resize "1200x webp q85" -}} {{- $processed = true -}} {{- end -}} {{- end -}} {{/* Attempt 2: Global assets */}} {{- if not $processed -}} {{- with $page.Params.featured_image -}} {{- $cleanPath := strings.TrimPrefix "/" . -}} {{- with resources.Get $cleanPath -}} {{- $img400 := .Resize "400x webp q75" -}} {{- $img800 := .Resize "800x webp q80" -}} {{- $img1200 := .Resize "1200x webp q85" -}} {{- $processed = true -}} {{- end -}} {{- end -}} {{- end -}} {{/* Fallback: External URL or non-processable — still add lazy loading */}} {{- if not $processed -}} {{- end -}} {{- end -}}
