{{/* Render Hook: Image — Injects lazy loading + async decoding on ALL markdown images. This catches every ![alt](src) in article content. */}} {{- $src := .Destination -}} {{- $alt := .Text | default "image" -}} {{- $title := .Title -}} {{/* Try to process via Hugo image pipeline */}} {{- $imgResource := false -}} {{- $cleanPath := strings.TrimPrefix "/" $src -}} {{- with resources.Get $cleanPath -}} {{- $imgResource = . -}} {{- end -}} {{- if not $imgResource -}} {{- with $.Page.Resources.GetMatch $src -}} {{- $imgResource = . -}} {{- end -}} {{- end -}} {{- if $imgResource -}} {{- $img800 := $imgResource.Resize "800x webp q80" -}} {{- else -}} {{- end -}}
