{{/* Render Hook: Image — Injects lazy loading + async decoding on ALL markdown images. This catches every ![alt](src) in article content. */}} {{- $src := .Destination -}} {{- $alt := .Text | default "image" -}} {{- $title := .Title -}} {{- $processed := false -}} {{/* Attempt 1: Global assets */}} {{- $cleanPath := strings.TrimPrefix "/" $src -}} {{- with resources.Get $cleanPath -}} {{- $img800 := .Resize "800x webp q80" -}} {{- $processed = true -}} {{- end -}} {{/* Attempt 2: Page bundle resource */}} {{- if not $processed -}} {{- with $.Page.Resources.GetMatch $src -}} {{- $img800 := .Resize "800x webp q80" -}} {{- $processed = true -}} {{- end -}} {{- end -}} {{/* Fallback: External or non-processable image */}} {{- if not $processed -}} {{- end -}}
